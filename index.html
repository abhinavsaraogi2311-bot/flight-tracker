// Flight Tracker — Google Apps Script Backend (CORS-safe version) // Paste into Extensions → Apps Script → Save → Deploy as Web App // Execute as: Me | Who has access: Anyone const SHEET_NAME = 'Flights'; const COLS = ['id', 'from', 'to', 'airline', 'date', 'flightNum', 'km']; function doGet(e) { const action = e.parameter.action; const sheet = getOrCreateSheet(); let result; try { if (action === 'read') { result = { ok: true, flights: readFlights(sheet) }; } else if (action === 'add') { const f = { id: e.parameter.id || String(Date.now()), from: e.parameter.from || '', to: e.parameter.to || '', airline: e.parameter.airline || '', date: e.parameter.date || '', flightNum: e.parameter.flightNum || '', km: Number(e.parameter.km) || 0 }; sheet.appendRow([f.id, f.from, f.to, f.airline, f.date, f.flightNum, f.km]); result = { ok: true, flight: f }; } else if (action === 'delete') { const id = e.parameter.id; deleteById(sheet, id); result = { ok: true }; } else { result = { ok: false, error: 'Unknown action: ' + action }; } } catch (err) { result = { ok: false, error: err.message }; } // CORS-safe: return JSON with permissive headers return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // ── Sheet helpers ───────────────────────────────────────────────────────────── function getOrCreateSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } // Ensure header if (sheet.getLastRow() === 0 || sheet.getRange(1, 1).getValue() !== 'id') { sheet.insertRowBefore(1); sheet.getRange(1, 1, 1, COLS.length).setValues([COLS]); } return sheet; } function readFlights(sheet) { const last = sheet.getLastRow(); if (last <= 1) return []; return sheet.getRange(2, 1, last - 1, COLS.length).getValues() .map(r => ({ id: String(r[0] || ''), from: String(r[1] || ''), to: String(r[2] || ''), airline: String(r[3] || ''), date: String(r[4] || ''), flightNum: String(r[5] || ''), km: Number(r[6]) || 0 })) .filter(f => f.from && f.to); } function deleteById(sheet, id) { const last = sheet.getLastRow(); if (last <= 1) return; const ids = sheet.getRange(2, 1, last - 1, 1).getValues().flat(); for (let i = ids.length - 1; i >= 0; i--) { if (String(ids[i]) === String(id)) { sheet.deleteRow(i + 2); // +2 because data starts at row 2 } } }
